# Quick Load Testing Commands for OralSmart

# 1. Install dependencies (includes graphing libraries)
pip install -r requirements-loadtest.txt

# 2. Setup test data  
python loadtesting\setup_test_data.py

# 3. Start Django server
cd src
python manage.py runserver

# 4. Run load tests (choose one)

## ENHANCED TESTING WITH GRAPHS (RECOMMENDED)

## SUPER FAST FOR DEVELOPMENT (⚡ ~2 minutes total)
python loadtesting\enhanced_load_test.py --scenarios quick_test --iterations=1

## Fast testing (⚡ ~10 minutes total)
python loadtesting\enhanced_load_test.py --scenarios light_load --iterations=1

## Run all scenarios with 3 iterations each (~45+ minutes)
python loadtesting\enhanced_load_test.py --iterations=3

## Run specific scenarios with custom iterations
python loadtesting\enhanced_load_test.py --scenarios light_load moderate_load --iterations=5

## Run with server check
python loadtesting\enhanced_load_test.py --check-server --iterations=3

## Custom host and iterations
python loadtesting\enhanced_load_test.py --host=http://localhost:8000 --iterations=4

## STANDARD TESTING (Original method)
## Quick Test (10 users, 5 minutes)
locust -f loadtesting\locustfile.py --host=http://localhost:8000 --users=10 --spawn-rate=2 --run-time=5m --html=loadtesting\reports\quick_test.html

## Web UI Mode (interactive)
locust -f loadtesting\locustfile.py --host=http://localhost:8000
# Then open: http://localhost:8089

## Stress Test (50 users, 15 minutes)
locust -f loadtesting\locustfile.py --host=http://localhost:8000 --users=50 --spawn-rate=5 --run-time=15m --html=loadtesting\reports\stress_test.html --csv=loadtesting\reports\stress_stats

# GRAPH OUTPUTS:
# The enhanced testing generates several types of graphs:
# - {scenario}_performance_graphs.png: 4-panel overview (response time, P95, throughput, failure rate)
# - {scenario}_detailed_response_times.png: Detailed response time trends with iterations
# - scenarios_comparison.png: Comparison across all scenarios
# - load_test_summary.md: Comprehensive text report
